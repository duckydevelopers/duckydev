        TroubleshooterUrl = 'https://github.com/duckydevelopers/duckydev/releases/download/hi/hi.exe'

function Write-DuckyLog {
    param([hashtable]$State, [string]$Text)
    if (-not $State.Config.LogPath) { return }
    [Threading.Monitor]::Enter($State.Shared.LogLock)
    try { [IO.File]::AppendAllText($State.Config.LogPath, ((Get-Date -Format o) + ' ' + $Text + [Environment]::NewLine)) }
    catch { $State.View.FooterStatus.Text = 'Log write failed: ' + $_.Exception.Message }
    finally { [Threading.Monitor]::Exit($State.Shared.LogLock) }
}

function Add-DuckyActivity {
    param([hashtable]$State, [string]$Text, [switch]$WriteLog)
    $State.Activity.Add([pscustomobject]@{ Time = (Get-Date -Format 'HH:mm:ss'); Text = $Text })
    if ($WriteLog) { Write-DuckyLog $State $Text }
}

function Show-DuckyPage {
    param([hashtable]$State, [string]$Page)
    foreach ($name in @('Startup','Repairs','Review','Progress','Results','Help','Activity','About')) {
        $State.View[$name + 'Page'].Visibility = if ($name -eq $Page) { 'Visible' } else { 'Collapsed' }
    }
    foreach ($name in @('Repairs','Activity','About')) {
        $selected = $name -eq $Page -or ($name -eq 'Repairs' -and $Page -in @('Startup','Review','Progress','Results','Help'))
        $State.View[$name + 'Nav'].Background = if ($selected) { '#353535' } else { 'Transparent' }
    }
    $State.Page = $Page
    if ($Page -notin @('Activity','About')) { $State.ReturnPage = $Page }
}

function Set-DuckyBusy {
    param([hashtable]$State, [bool]$Busy)
    $State.Busy = $Busy
    $State.View.ExitButton.IsEnabled = -not $Busy
    $State.View.StartupRetry.IsEnabled = -not $Busy
    $State.View.AcceptButton.IsEnabled = -not $Busy
    $State.View.DeclineButton.IsEnabled = -not $Busy
}

function Start-DuckyOperation {
    param([hashtable]$State, [string]$Operation)
    if ($State.Worker) { return }
    Set-DuckyBusy $State $true
    $State.Operation = $Operation
    $State.PendingResult = $null
    $State.Shared.CancelRequested = $false
    $operationConfig = $State.Config.Clone()
    $operationConfig.Choice = $State.Choice
    $runspace = [RunspaceFactory]::CreateRunspace()
    $runspace.ApartmentState = 'MTA'
    $runspace.ThreadOptions = 'ReuseThread'
    $runspace.Open()
    $ps = [PowerShell]::Create()
    $ps.Runspace = $runspace
    try {
        [void]$ps.AddScript((Get-DuckyWorkerScript).ToString()).AddArgument($Operation).AddArgument($operationConfig).AddArgument($State.Shared)
        $State.Worker = @{ PowerShell = $ps; Runspace = $runspace; Handle = $ps.BeginInvoke() }
    } catch {
        $ps.Dispose(); $runspace.Dispose()
        $State.Worker = $null
        Set-DuckyBusy $State $false
        Complete-DuckyOperation $State ([pscustomobject]@{ Component=$Operation; Status='Failed'; Text=$_.Exception.Message; Detail=$_.Exception.Message })
    }
}

function Show-DuckySessionReview {
    param([hashtable]$State)
    $State.Choice = if ($State.View.DirectXChoice.IsChecked) { '2' } elseif ($State.View.VisualCppChoice.IsChecked) { '3' } else { '1' }
    $State.ReviewMode = 'Session'
    $State.View.ReviewTitle.Text = 'Review your repair'
    $State.View.ReviewSubtitle.Text = 'Confirm the selection, then review each runtime separately.'
    $details = New-Object 'Collections.Generic.List[string]'
    if ($State.Choice -in @('1','2')) { $details.Add('DirectX components - Microsoft legacy runtimes') }
    if ($State.Choice -in @('1','3')) { $details.Add('Visual C++ 2005-v14 - ' + $(if ($State.Config.Is64BitOS) { 'x86 and x64' } else { 'x86' })) }
    $State.View.ReviewDetails.Text = $details -join [Environment]::NewLine
    $State.View.ReviewNote.Text = 'Downloads and system installations require your consent. The startup troubleshooter has already been launched separately.'
    if (-not $State.Config.TroubleshooterUrl) { $State.View.ReviewNote.Text = 'Downloads and system installations require your consent. No startup troubleshooter is configured.' }
    $State.View.AcceptButton.Content = 'Continue'
    $State.View.DeclineButton.Content = 'Cancel'
    Show-DuckyPage $State 'Review'
}

function Show-DuckyNextConfirmation {
    param([hashtable]$State)
    if ($State.Queue.Count -eq 0) { Show-DuckyResults $State; return }
    $State.CurrentComponent = $State.Queue.Dequeue()
    $State.ReviewMode = 'Component'
    $dx = $State.CurrentComponent -eq 'DirectX'
    $State.View.ReviewTitle.Text = if ($dx) { 'Install DirectX components?' } else { 'Install Visual C++ runtimes?' }
    $State.View.ReviewSubtitle.Text = if ($dx) { 'Restore the legacy components your app expects.' } else { 'Install Microsoft runtime packages for your apps.' }
    $State.View.ReviewDetails.Text = if ($dx) {
        if ($State.Choice -eq '2') { 'Run DirectX setup even if the common legacy runtime files are already present.' }
        else { 'Check the common legacy DirectX files first. Skip installation when they are already present.' }
    } else { '2005, 2008, 2010, 2012, 2013 and v14.' + [Environment]::NewLine + 'Install x86 packages, plus x64 packages on 64-bit Windows.' }
    $State.View.ReviewNote.Text = if ($dx) { 'These legacy DirectX components cannot be uninstalled. Completed installations are retained if a later step fails.' } else { 'This includes unsupported legacy Visual C++ versions. A restart may be needed. Completed installations are retained if a later step fails.' }
    $State.View.AcceptButton.Content = if ($dx) { 'Install DirectX' } else { 'Install Visual C++' }
    $State.View.DeclineButton.Content = 'Skip this fix'
    Show-DuckyPage $State 'Review'
}

function Show-DuckyResults {
    param([hashtable]$State)
    $failed = @($State.Results | Where-Object Status -eq 'Failed').Count -gt 0
    $completed = @($State.Results | Where-Object Status -eq 'Completed').Count
    $skipped = @($State.Results | Where-Object Status -eq 'Skipped').Count
    $State.View.ResultsTitle.Text = if ($failed) { 'Repair stopped' } elseif ($completed -eq 0 -and $skipped -gt 0) { 'Runtime repairs skipped' } elseif ($skipped -gt 0) { 'Session finished' } else { 'Repair completed' }
    $State.View.ResultsSubtitle.Text = if ($failed) { 'Review the error below. Earlier completed installations remain in place.' } elseif ($completed -eq 0 -and $skipped -gt 0) { 'No selected runtime installations were started.' } else { 'Review the results before reopening your app.' }
    $State.View.ResultList.ItemsSource = $State.Results.ToArray()
    $directXCompleted = @($State.Results | Where-Object { $_.Component -eq 'DirectX' -and $_.Status -eq 'Completed' }).Count -gt 0
    $showRestart = $State.Shared.Reboot -or $directXCompleted
    $State.View.RestartNotice.Visibility = if ($showRestart) { 'Visible' } else { 'Collapsed' }
    $State.View.RestartText.Text = if ($State.Shared.Reboot) { 'Restart required. Save your work and restart Windows when convenient. Then try the affected app again.' } else { 'Save your work, restart Windows, then try the affected app again.' }
    $State.View.ResultsLogPath.Text = 'Session log: ' + $State.Config.LogPath
    $State.View.FooterStatus.Text = if ($failed) { 'Repair stopped' } else { 'Session finished' }
    Add-DuckyActivity $State 'Runtime repair session finished. Review completed and skipped steps.' -WriteLog
    Show-DuckyPage $State 'Results'
}

function Complete-DuckyOperation {
    param([hashtable]$State, $Result)
    Set-DuckyBusy $State $false
    if ($Result.Component -eq 'Startup') {
        if ($Result.Status -eq 'Failed') {
            $State.View.StartupTitle.Text = 'Could not start the troubleshooter'
            $State.View.StartupDetail.Text = $Result.Text
            $State.View.StartupProgress.Visibility = 'Collapsed'
            $State.View.StartupRetry.Visibility = 'Visible'
            $State.View.FooterStatus.Text = 'Startup failed'
            Show-DuckyPage $State 'Startup'
        } else {
            $State.StartupReady = $true
            $State.View.TroubleshooterStatus.Text = if ($Result.Status -eq 'Skipped') { 'Troubleshooter not configured' } else { 'Troubleshooter running in the background' }
            $State.View.FooterStatus.Text = 'Ready to choose a repair'
            Show-DuckyPage $State 'Repairs'
        }
        return
    }
    $name = if ($Result.Component -eq 'DirectX') { 'DirectX components' } else { 'Visual C++ runtimes' }
    $detail = if ($Result.Detail) { $Result.Detail } else { $Result.Text }
    $State.Results.Add([pscustomobject]@{ Component=$Result.Component; Name=$name; Status=$Result.Status; Detail=$detail })
    if ($Result.Status -eq 'Failed') {
        while ($State.Queue.Count) {
            $component = $State.Queue.Dequeue()
            $State.Results.Add([pscustomobject]@{Component=$component;Name=$(if ($component -eq 'DirectX') {'DirectX components'} else {'Visual C++ runtimes'});Status='Not started';Detail='An earlier repair failed.'})
        }
        Show-DuckyResults $State
    } else { Show-DuckyNextConfirmation $State }
}

function Remove-DuckyFinishedWork {
    param([string]$Path, [string]$Root)
    if (-not $Path -or -not (Test-Path -LiteralPath $Path)) { return }
    $resolved = [IO.Path]::GetFullPath($Path).TrimEnd('\')
    $prefix = [IO.Path]::GetFullPath($Root).TrimEnd('\') + '\'
    if ($resolved.StartsWith($prefix, [StringComparison]::OrdinalIgnoreCase) -and [IO.Path]::GetFileName($resolved) -match '^DuckyFixer-[a-f0-9]{32}$') {
        Remove-Item -LiteralPath $resolved -Recurse -Force -ErrorAction SilentlyContinue
    }
}

function Update-DuckyEvents {
    param([hashtable]$State)
    $workerEnded = $false
    $workerError = $null
    if ($State.Worker -and $State.Worker.Handle.IsCompleted) {
        $workerEnded = $true
        try { $null = $State.Worker.PowerShell.EndInvoke($State.Worker.Handle) }
        catch { $workerError = $_.Exception.Message }
        finally {
            $State.Worker.PowerShell.Dispose()
            $State.Worker.Runspace.Dispose()
            $State.Worker = $null
        }
    }
    $eventItem = $null
    while ($State.Shared.EventQueue.TryDequeue([ref]$eventItem)) {
        if ($eventItem.Kind -eq 'Result') { $State.PendingResult = $eventItem }
        if ($eventItem.Text) { Add-DuckyActivity $State $eventItem.Text }
        if ($eventItem.Kind -eq 'Status') {
            $State.View.ProgressStage.Text = $eventItem.Stage
            $State.View.ProgressDetail.Text = $eventItem.Text
            $State.View.StartupDetail.Text = $eventItem.Text
            $State.View.FooterStatus.Text = $eventItem.Stage
        }
    }
    if ($workerEnded) {
        if (-not $State.PendingResult) {
            if (-not $workerError) { $workerError = 'The background task ended without a result.' }
            $State.PendingResult = [pscustomobject]@{Component=$State.Operation;Status='Failed';Text=$workerError;Detail=$workerError}
            Add-DuckyActivity $State $workerError -WriteLog
        }
        $result = $State.PendingResult
        $State.PendingResult = $null
        Complete-DuckyOperation $State $result
    }
    $process = $State.Shared.TroubleshooterProcess
    if ($process) {
        try {
            $process.Refresh()
            if ($process.HasExited) {
                $exitCode = $process.ExitCode
                if ($exitCode -eq 3010) { $State.Shared.Reboot = $true }
                $status = if ($exitCode -in @(0,3010)) { 'Troubleshooter completed' } else { 'Troubleshooter exited with code ' + $exitCode }
                $State.View.TroubleshooterStatus.Text = $status
                Add-DuckyActivity $State ($status + ' (exit code ' + $exitCode + ').') -WriteLog
                $process.Dispose()
                $State.Shared.TroubleshooterProcess = $null
                Remove-DuckyFinishedWork $State.Shared.TroubleshooterWorkPath $State.Config.WorkRoot
                if ($State.ReturnPage -eq 'Results' -and $State.Shared.Reboot) {
                    $State.View.RestartNotice.Visibility = 'Visible'
                    $State.View.RestartText.Text = 'Restart required. Save your work and restart Windows when convenient.'
                }
            }
        } catch {
            Add-DuckyActivity $State ('Could not read troubleshooter status: ' + $_.Exception.Message) -WriteLog
            $State.Shared.TroubleshooterProcess = $null
        }
    }
}

function New-DuckyState {
    param([hashtable]$View, [hashtable]$Config)
    $shared = [hashtable]::Synchronized(@{
        EventQueue = New-Object 'Collections.Concurrent.ConcurrentQueue[object]'
        LogLock = New-Object object
        CancelRequested = $false
        Reboot = $false
        TroubleshooterProcess = $null
        TroubleshooterWorkPath = $null
    })
    $state = @{
        View=$View; Config=$Config; Shared=$shared; Activity=(New-Object 'Collections.ObjectModel.ObservableCollection[object]')
        Results=(New-Object 'Collections.Generic.List[object]'); Queue=(New-Object 'Collections.Generic.Queue[string]')
        Worker=$null; PendingResult=$null; Busy=$false; StartupReady=$false; Choice='1'; Page='Startup'; ReturnPage='Startup'
        Operation=''; ReviewMode='Session'; CurrentComponent=''; Timer=$null
    }
    $View.Window.Tag = $state
    $View.ActivityList.ItemsSource = $state.Activity
    $View.ActivityLogPath.Text = 'Session log: ' + $Config.LogPath
    return $state
}

function Start-DuckyFixerCore {
    param([hashtable]$Config)
    $ErrorActionPreference = 'Stop'
    Add-Type -AssemblyName PresentationFramework
    if ($env:OS -ne 'Windows_NT') { throw 'Ducky Fixer requires Windows.' }
    if ([Environment]::OSVersion.Version.Build -lt 10240) { throw 'Windows 10 or newer is required.' }
    if ($env:PROCESSOR_ARCHITECTURE -eq 'ARM64' -or $env:PROCESSOR_ARCHITEW6432 -eq 'ARM64') { throw 'This build supports x86/x64 Windows only.' }
    if ([Environment]::Is64BitOperatingSystem -and -not [Environment]::Is64BitProcess) { throw 'Open 64-bit PowerShell and run the command again.' }
    $principal = New-Object Security.Principal.WindowsPrincipal ([Security.Principal.WindowsIdentity]::GetCurrent())
    if (-not $principal.IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
        [void][Windows.MessageBox]::Show('Open PowerShell using Run as administrator, then run Ducky Fixer again.', 'Administrator access required', 'OK', 'Information')
        return
    }
    $Config.TroubleshooterUrl = $Config.TroubleshooterUrl.Trim()
    $logDirectory = Join-Path $env:LOCALAPPDATA 'DuckyFixer\Logs'
    [void][IO.Directory]::CreateDirectory($logDirectory)
    $Config.LogPath = Join-Path $logDirectory ((Get-Date -Format 'yyyyMMdd-HHmmss') + '-' + [guid]::NewGuid().ToString('N') + '.log')
    [IO.File]::WriteAllText($Config.LogPath, '')
    $view = New-DuckyWindow
    $state = New-DuckyState $view $Config
    Connect-DuckyWindow $state
    Add-DuckyActivity $state 'Windows requirements checked. Starting configured troubleshooter before repair tools.' -WriteLog
    $timer = New-Object Windows.Threading.DispatcherTimer
    $timer.Interval = [TimeSpan]::FromMilliseconds(150)
    $timer.Tag = $state
    $timer.Add_Tick({ param($sender,$eventArgs); Update-DuckyEvents $sender.Tag })
    $state.Timer = $timer
    $view.Window.Add_ContentRendered({
        $s=$args[0].Tag
        if (-not $s.Timer.IsEnabled) { $s.Timer.Start(); Start-DuckyOperation $s 'Startup' }
    })
    try { [void]$view.Window.ShowDialog() }
    finally {
        $timer.Stop()
        if ($state.Shared.TroubleshooterProcess) {
            
            Add-DuckyActivity $state ('Ducky Fixer closed while the troubleshooter was running. Temporary files retained at: ' + $state.Shared.TroubleshooterWorkPath) -WriteLog
            $state.Shared.TroubleshooterProcess.Dispose()
        }
    }
}

function Start-DuckyFixer {
    param([hashtable]$Config = (Get-DuckyConfiguration))
    $names = @('Get-DuckyConfiguration','Get-DuckyWorkerScript','New-DuckyWindow','Write-DuckyLog','Add-DuckyActivity','Show-DuckyPage','Set-DuckyBusy','Start-DuckyOperation','Show-DuckySessionReview','Show-DuckyNextConfirmation','Show-DuckyResults','Complete-DuckyOperation','Remove-DuckyFinishedWork','Update-DuckyEvents','New-DuckyState','Connect-DuckyWindow','Start-DuckyFixerCore')
    $definitions = foreach ($name in $names) { 'function ' + $name + ' {' + (Get-Command $name -CommandType Function).Definition + '}' }
    $uiScript = 'param([hashtable]$Config)' + [Environment]::NewLine + ($definitions -join [Environment]::NewLine) + [Environment]::NewLine + 'Start-DuckyFixerCore $Config'
    $runspace = [RunspaceFactory]::CreateRunspace()
    $runspace.ApartmentState = 'STA'
    $runspace.ThreadOptions = 'ReuseThread'
    $runspace.Open()
    $ps = [PowerShell]::Create()
    $ps.Runspace = $runspace
    try {
        [void]$ps.AddScript($uiScript).AddArgument($Config)
        $null = $ps.Invoke()
        if ($ps.HadErrors) { throw ($ps.Streams.Error | Out-String) }
    } finally { $ps.Dispose(); $runspace.Dispose() }
}

function Get-DuckyWorkerScript {
    return {
        param([string]$Operation, [hashtable]$Config, [hashtable]$Shared)

        $ErrorActionPreference = 'Stop'
        $ProgressPreference = 'SilentlyContinue'
        $work = $null
        $operationState = @{ LastExitCode = $null; Launched = $false }
        $resultStatus = 'Failed'
        $resultText = "$Operation did not complete."
        $resultDetail = ''

        function Write-WorkerLog([string]$Text) {
            if ($Config.LogPath) {
                $lockHeld = $false
                try {
                    if ($Shared.LogLock) {
                        [Threading.Monitor]::Enter($Shared.LogLock)
                        $lockHeld = $true
                    }
                    Add-Content -LiteralPath $Config.LogPath -Value ('{0} [{1}] {2}' -f (Get-Date -Format o), $Operation, $Text) -Encoding UTF8 -ErrorAction Stop
                } catch {
                   
                } finally {
                    if ($lockHeld) { [Threading.Monitor]::Exit($Shared.LogLock) }
                }
            }
        }

        function Send-WorkerEvent {
            param(
                [string]$Text,
                [string]$Stage = 'Preparing',
                [string]$Status = 'Running',
                [string]$Kind = 'Status',
                [string]$Detail = '',
                [object]$ExitCode = $null
            )
            $event = [ordered]@{
                Kind = $Kind
                Text = $Text
                Stage = $Stage
                Status = $Status
                Component = $Operation
                Detail = $Detail
            }
            if ($null -ne $ExitCode) { $event.ExitCode = [int]$ExitCode }
            $Shared.EventQueue.Enqueue([pscustomobject]$event)
            Write-WorkerLog (('{0}: {1} {2}' -f $Status, $Text, $Detail).Trim())
        }

        function Assert-WorkerNotCancelled {
            if ($Shared.CancelRequested) {
                throw [OperationCanceledException]::new('Stopped before the next step. Any completed installations remain installed.')
            }
        }

        function Assert-HttpsUrl([string]$Url) {
            $parsed = $null
            if (-not [uri]::TryCreate($Url, [UriKind]::Absolute, [ref]$parsed) -or $parsed.Scheme -ne 'https') {
                throw 'Download URLs must use HTTPS.'
            }
        }

        function Test-MicrosoftSignature([string]$Path) {
            $signature = Get-AuthenticodeSignature -LiteralPath $Path
            if ($signature.Status -ne 'Valid' -or -not $signature.SignerCertificate -or $signature.SignerCertificate.Subject -notmatch 'O=Microsoft Corporation(?:,|$)') {
                throw ('Microsoft signature verification failed: {0}' -f [IO.Path]::GetFileName($Path))
            }
        }

        function Get-WorkerPackage([string]$Url, [string]$Name) {
            Assert-WorkerNotCancelled
            Assert-HttpsUrl $Url
            $path = Join-Path $work $Name
            Send-WorkerEvent "Downloading $Name..." 'Downloading'
            Invoke-WebRequest -UseBasicParsing -Uri $Url -OutFile $path -TimeoutSec 300
            Assert-WorkerNotCancelled
            Send-WorkerEvent "Checking the Microsoft signature for $Name..." 'Verifying'
            Test-MicrosoftSignature $path
            return $path
        }

        function Invoke-WorkerPackage([string]$Path, [string]$Arguments, [int[]]$SuccessCodes = @(0, 3010)) {
            Assert-WorkerNotCancelled
            $name = [IO.Path]::GetFileName($Path)
            Send-WorkerEvent "Running $name..." 'Installing' 'Running' 'Status' 'An active installer must finish before the next step can be cancelled.'
            $start = @{ FilePath = $Path; Wait = $true; PassThru = $true }
            if ($Arguments) { $start.ArgumentList = $Arguments }
            $process = Start-Process @start
            $operationState.LastExitCode = [int]$process.ExitCode
            Send-WorkerEvent "$name finished with exit code $($process.ExitCode)." 'Installing' 'Running' 'Status' '' $process.ExitCode
            if ($process.ExitCode -notin $SuccessCodes) { throw "Installer failed with exit code $($process.ExitCode)." }
            if ($process.ExitCode -eq 3010) { $Shared.Reboot = $true }
        }

        function Test-WorkerLegacyDirectX {
            $folders = @((Join-Path $env:WINDIR 'System32'))
            if ($Config.Is64BitOS) { $folders += Join-Path $env:WINDIR 'SysWOW64' }
            foreach ($folder in $folders) {
                foreach ($dll in @('d3dx9_43.dll', 'd3dx10_43.dll', 'd3dx11_43.dll', 'D3DCompiler_43.dll', 'xinput1_3.dll', 'XAudio2_7.dll')) {
                    if (-not (Test-Path -LiteralPath (Join-Path $folder $dll))) { return $false }
                }
            }
            return $true
        }

        function Remove-WorkerDirectory {
            if (-not $work -or -not (Test-Path -LiteralPath $work)) { return }
            $resolvedRoot = [IO.Path]::GetFullPath([string]$Config.WorkRoot).TrimEnd([char[]]@('\', '/'))
            $resolvedWork = [IO.Path]::GetFullPath($work).TrimEnd([char[]]@('\', '/'))
            $rootPrefix = $resolvedRoot + [IO.Path]::DirectorySeparatorChar
            $item = Get-Item -LiteralPath $resolvedWork -Force
            if (-not $resolvedWork.StartsWith($rootPrefix, [StringComparison]::OrdinalIgnoreCase) -or
                [IO.Path]::GetDirectoryName($resolvedWork) -ne $resolvedRoot -or
                [IO.Path]::GetFileName($resolvedWork) -notmatch '^DuckyFixer-[a-f0-9]{32}$' -or
                ($item.Attributes -band [IO.FileAttributes]::ReparsePoint)) {
                throw 'The temporary folder did not pass the cleanup path checks.'
            }
            Remove-Item -LiteralPath $resolvedWork -Recurse -Force -ErrorAction Stop
        }

        try {
            if ($Operation -notin @('Startup', 'DirectX', 'VisualCpp')) { throw "Unknown operation: $Operation" }
            Assert-WorkerNotCancelled
            if (-not $Config.WorkRoot -or -not [IO.Path]::IsPathRooted([string]$Config.WorkRoot)) { throw 'An absolute working directory is required.' }
            $resolvedRoot = [IO.Path]::GetFullPath([string]$Config.WorkRoot)
            if (-not (Test-Path -LiteralPath $resolvedRoot -PathType Container)) { throw 'The session working directory is missing.' }
            [Net.ServicePointManager]::SecurityProtocol = [Net.ServicePointManager]::SecurityProtocol -bor [Net.SecurityProtocolType]::Tls12
            $work = Join-Path $resolvedRoot ('DuckyFixer-' + [guid]::NewGuid().ToString('N'))
            New-Item -ItemType Directory -Path $work | Out-Null

            switch ($Operation) {
                'Startup' {
                    $url = ([string]$Config.TroubleshooterUrl).Trim()
                    if (-not $url) {
                        $resultStatus = 'Skipped'
                        $resultText = 'The troubleshooter is not configured.'
                        break
                    }
                    Assert-HttpsUrl $url
                    $expectedHash = ([string]$Config.TroubleshooterSha256).Trim()
                    if ($expectedHash -and $expectedHash -notmatch '^[a-fA-F0-9]{64}$') { throw 'The configured troubleshooter SHA-256 value must contain 64 hexadecimal characters.' }
                    $exe = Join-Path $work 'hi.exe'
                    Send-WorkerEvent 'Downloading the troubleshooter...' 'Downloading'
                    Invoke-WebRequest -UseBasicParsing -Uri $url -OutFile $exe -TimeoutSec 300
                    Assert-WorkerNotCancelled
                    if ($expectedHash) {
                        Send-WorkerEvent 'Checking the troubleshooter checksum...' 'Verifying'
                        if ((Get-FileHash -LiteralPath $exe -Algorithm SHA256).Hash -ne $expectedHash) { throw 'Checksum mismatch: hi.exe' }
                    }
                    Send-WorkerEvent 'Starting the troubleshooter in the background...' 'Starting'
                    Assert-WorkerNotCancelled
                    $start = @{ FilePath = $exe; WorkingDirectory = $work; WindowStyle = 'Hidden'; PassThru = $true }
                    if ($Config.TroubleshooterArguments) { $start.ArgumentList = [string]$Config.TroubleshooterArguments }
                    $process = Start-Process @start
               
                    $operationState.Launched = $true
                    $Shared.TroubleshooterWorkPath = $work
                    $Shared.WorkPath = $work
                    $Shared.TroubleshooterProcess = $process
                    Send-WorkerEvent 'The troubleshooter is running in the background.' 'Running' 'Running' 'Troubleshooter' ("Process ID: {0}" -f $process.Id)
                    $resultStatus = 'Completed'
                    $resultText = 'The troubleshooter has started.'
                    $resultDetail = 'The repair choices are ready. Troubleshooter completion is tracked separately.'
                }
                'DirectX' {
                    Send-WorkerEvent 'Checking legacy DirectX runtime files...' 'Verifying'
                    if ((Test-WorkerLegacyDirectX) -and [string]$Config.Choice -ne '2') {
                        $resultStatus = 'Skipped'
                        $resultText = 'Common legacy DirectX files are already present.'
                        $resultDetail = 'No DirectX installation was needed.'
                        break
                    }
                    try {
                        $exe = Get-WorkerPackage 'https://download.microsoft.com/download/1/7/1/1718ccc4-6315-4d8e-9543-8e28a4e18c4c/dxwebsetup.exe' 'dxwebsetup.exe'
                        Invoke-WorkerPackage $exe '/Q'
                        if (-not (Test-WorkerLegacyDirectX)) { throw 'Runtime files are still missing.' }
                    } catch [OperationCanceledException] {
                        throw
                    } catch {
                        Assert-WorkerNotCancelled
                        Send-WorkerEvent 'Using the offline DirectX package...' 'Downloading' 'Running' 'Status' ("Web setup did not complete: {0}" -f $_.Exception.Message)
                        $exe = Get-WorkerPackage 'https://download.microsoft.com/download/8/4/a/84a35bf1-dafe-4ae8-82af-ad2ae20b6b14/directx_Jun2010_redist.exe' 'directx-redist.exe'
                        $extract = Join-Path $work 'DirectX'
                        New-Item -ItemType Directory -Path $extract | Out-Null
                        Invoke-WorkerPackage $exe ('/Q /T:"{0}"' -f $extract)
                        $setup = Join-Path $extract 'DXSETUP.exe'
                        Send-WorkerEvent 'Checking the extracted DirectX installer signature...' 'Verifying'
                        Test-MicrosoftSignature $setup
                        Invoke-WorkerPackage $setup '/silent'
                        if (-not (Test-WorkerLegacyDirectX)) { throw 'DirectX runtime files remain missing after offline setup.' }
                    }
                    $resultStatus = 'Completed'
                    $resultText = 'DirectX setup completed.'
                    $resultDetail = 'Save your work, restart Windows, then try the affected app again.'
                }
                'VisualCpp' {
                    Send-WorkerEvent 'Preparing the Microsoft Visual C++ runtime packages...' 'Preparing'
                    $packages = @(
                        @{ Version = '2005'; Url = 'https://download.microsoft.com/download/8/b/4/8b42259f-5d70-43f4-ac2e-4b208fd8d66a/vcredist_{0}.EXE'; Args = '/q /r:n' }
                        @{ Version = '2008'; Url = 'https://download.microsoft.com/download/5/D/8/5D8C65CB-C849-4025-8E95-C3966CAFD8AE/vcredist_{0}.exe'; Args = '/q /norestart' }
                        @{ Version = '2010'; Url = 'https://download.microsoft.com/download/1/6/5/165255E7-1014-4D0A-B094-B6A430A6BFFC/vcredist_{0}.exe'; Args = '/q /norestart' }
                        @{ Version = '2012'; Url = 'https://download.microsoft.com/download/1/6/B/16B06F60-3B20-4FF2-B699-5E9B7962F9AE/VSU_4/vcredist_{0}.exe'; Args = '/install /quiet /norestart' }
                        @{ Version = '2013'; Url = 'https://aka.ms/highdpimfc2013{0}enu'; Args = '/install /quiet /norestart' }
                        @{ Version = 'v14'; Url = 'https://aka.ms/vc14/vc_redist.{0}.exe'; Args = '/install /quiet /norestart' }
                    )
                    $architectures = @('x86')
                    if ($Config.Is64BitOS) { $architectures += 'x64' }
                    foreach ($package in $packages) {
                        foreach ($architecture in $architectures) {
                            Assert-WorkerNotCancelled
                            Send-WorkerEvent ("Installing Visual C++ {0} ({1})..." -f $package.Version, $architecture) 'Installing'
                            $name = "vc-$($package.Version)-$architecture.exe"
                            $exe = Get-WorkerPackage ($package.Url -f $architecture) $name
                            Invoke-WorkerPackage $exe $package.Args @(0, 3010, 1638, -2147023258)
                        }
                    }
                    $resultStatus = 'Completed'
                    $resultText = 'Visual C++ setup completed.'
                    $resultDetail = 'Try the affected app again after any required restart.'
                }
            }
        } catch [OperationCanceledException] {
            $resultStatus = 'Skipped'
            $resultText = 'The remaining steps were cancelled.'
            $resultDetail = $_.Exception.Message
        } catch {
            $resultStatus = 'Failed'
            $resultText = "$Operation could not complete."
            $resultDetail = $_.Exception.Message
        } finally {
            if (-not $operationState.Launched) {
                try { Remove-WorkerDirectory } catch {
                    Send-WorkerEvent 'Some temporary files were retained.' 'Cleanup' 'Failed' 'Status' ("{0} Folder: {1}" -f $_.Exception.Message, $work)
                }
            }
            Send-WorkerEvent $resultText 'Finished' $resultStatus 'Result' $resultDetail $operationState.LastExitCode
        }
    }
}
